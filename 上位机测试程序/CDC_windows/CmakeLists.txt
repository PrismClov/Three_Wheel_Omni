cmake_minimum_required(VERSION 3.10)
project(MySerialApp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CSERIALPORT_ENABLE_NATIVE_SYNC ON CACHE BOOL "" FORCE)
set(CSERIALPORT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CSERIALPORT_BUILD_BINDING_C OFF CACHE BOOL "" FORCE)
set(CSERIALPORT_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(CSERIALPORT_BUILD_DOC OFF CACHE BOOL "" FORCE)

add_subdirectory(thirdparty/CSerialPort)

add_executable(MySerialApp main.cpp)

target_include_directories(MySerialApp PRIVATE
    ${CMAKE_SOURCE_DIR}/thirdparty/CSerialPort/include
)

target_link_libraries(MySerialApp PRIVATE
    libcserialport
)
